using UnityEngine; public class DatingSystem : MonoBehaviour { public string playerName = "Lina"; public int datingPrice = 500; public int affection = 0; public bool married = false; public void SetPrice(int price) { datingPrice = Mathf.Max(0, price); Debug.Log("السعر المحدد: " + datingPrice); } public void AcceptDate(string characterName, int offer) { if (offer >= datingPrice) { affection += 20; Debug.Log(characterName + " وافق على الموعد مع " + playerName); } else { Debug.Log("العرض أقل من السعر المحدد."); } } public void AcceptMarriage(string characterName) { if (affection >= 50) { married = true; Debug.Log("تم الزواج من " + characterName + " ❤️"); } else { Debug.Log("تحتاجان إلى بناء علاقة أقوى أولاً."); } } public void RomanticInteraction(string characterName) { if (married) { affection += 10; Debug.Log("لحظة رومانسية جميلة مع " + characterName + " 💕"); } } }
