import tkinter as tk from tkinter import ttk import random import string import requests import threading from concurrent.futures import ThreadPoolExecutor # ===================================================== # ✨ إعدادات الألوان والثيم الفاخر ✨ # ===================================================== BG = "#07101f" PANEL = "#0d1a2c" PANEL2 = "#101f34" PINK = "#ff1493" TEXT = "#f3f6ff" MUTED = "#8ea0b8" GREEN = "#29d391" RED = "#ff4d6d" # ===================================================== # ✨ النافذة الرئيسية ✨ # ===================================================== root = tk.Tk() root.title("RwL • Username Lab & Auto Diagnostics") root.geometry("1100x720") root.configure(bg=BG) style = ttk.Style() style.theme_use("clam") # الهيدر الفاخر title = tk.Label( root, text="♛ RwL", font=("Arial", 34, "bold"), bg=BG, fg=TEXT ) title.pack(pady=(22, 0)) sub = tk.Label( root, text="AUTOMATED USERNAME LAB • Generate • Auto-Check • Save", font=("Arial", 11), bg=BG, fg=MUTED ) sub.pack(pady=(0, 18)) main = tk.Frame(root, bg=BG) main.pack(fill="both", expand=True, padx=28, pady=10) # القائمة اليسرى (التحكم) left = tk.Frame( main, bg=PANEL, highlightbackground="#20324b", highlightthickness=1 ) left.pack(side="left", fill="y", padx=(0, 12)) left.configure(width=430) left.pack_propagate(False) # القائمة اليمنى (النتائج) right = tk.Frame( main, bg=PANEL, highlightbackground="#20324b", highlightthickness=1 ) right.pack(side="left", fill="both", expand=True) tk.Label( left, text="TikTok System", font=("Arial", 16, "bold"), bg=PANEL, fg=TEXT ).pack(anchor="w", padx=24, pady=(24, 5)) tk.Label( left, text="مولّد وفاحص تلقائي لأسماء المستخدمين", font=("Arial", 10), bg=PANEL, fg=MUTED ).pack(anchor="w", padx=24) # نمط الإنشاء tk.Label( left, text="طريقة الإنشاء", bg=PANEL, fg=MUTED, font=("Arial", 10) ).pack(anchor="w", padx=24, pady=(20, 6)) mode = ttk.Combobox( left, values=["تلقائي", "بادئة + أرقام", "حروف + أرقام"], state="readonly" ) mode.current(0) mode.pack(fill="x", padx=24) # عدد اليوزرات tk.Label( left, text="عدد اليوزرات", bg=PANEL, fg=MUTED, font=("Arial", 10) ).pack(anchor="w", padx=24, pady=(14, 6)) count_var = tk.IntVar(value=20) count = tk.Spinbox( left, from_=1, to=1000, textvariable=count_var, bg=PANEL2, fg=TEXT, buttonbackground=PANEL2, insertbackground=TEXT, relief="flat" ) count.pack(fill="x", padx=24) # البادئة tk.Label( left, text="البادئة (اختياري)", bg=PANEL, fg=MUTED, font=("Arial", 10) ).pack(anchor="w", padx=24, pady=(14, 6)) prefix_var = tk.StringVar() tk.Entry( left, textvariable=prefix_var, bg=PANEL2, fg=TEXT, insertbackground=TEXT, relief="flat" ).pack(fill="x", padx=24) # Variable لمتابعة حالة الفحص is_checking = False # ===================================================== # ✨ منطق البرمجة المدمج (توليد وفحص تلقائي) ✨ # ===================================================== def make_username(): prefix = prefix_var.get().strip().lower() mode_value = mode.get() if mode_value == "بادئة + أرقام": return (prefix or "rwl") + "".join(random.choices(string.digits, k=4)) if mode_value == "حروف + أرقام": chars = string.ascii_lowercase + string.digits return "".join(random.choices(chars, k=6)) words = ["rwl", "nova", "sara", "moon", "vibe", "queen", "star"] return random.choice(words) + str(random.randint(10, 9999)) def valid_username(name): return 2 <= len(name) <= 24 and all(c.isalnum() or c in "._" for c in name) def check_tiktok_user_live(username): """ دالة الفحص التلقائي الحقيقي عبر طلب HTTP للمنصة """ url = f"https://www.tiktok.com/@{username}" headers = { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/115.0.0.0 Safari/537.36" } try: response = requests.get(url, headers=headers, timeout=5) # إذا كانت الاستجابة 404 فغالباً اليوزر غير موجود ومتاح للتسجيل if response.status_code == 404: return True return False except Exception: # في حال حدوث خطأ اتصال يعتبر الحساب غير محدد أو غير متاح return False def save_available_user(username): """حفظ تلقائي لليوزرات المتاحة""" with open("available_users.txt", "a", encoding="utf-8") as f: f.write(f"@{username}\n") def process_single_user(u): """فحص يوزر واحد وتحديث واجهة المستخدم فوراً""" global available_count, unavailable_count is_avail = check_tiktok_user_live(u) if is_avail: status = "متاح تلقائياً" tag = "ok" save_available_user(u) else: status = "غير متاح" tag = "no" # تحديث الواجهة التفاعلية بأمان في الخيط الرئيسي def update_ui(): global available_count, unavailable_count tree.insert("", "end", values=("@" + u, status, "نسخ"), tags=(tag,)) if is_avail: available_count += 1 available_var.set(str(available_count)) else: unavailable_count += 1 unavailable_var.set(str(unavailable_count)) root.after(0, update_ui) def run_auto_diagnostics_thread(): """خيط عمل مستقل لمنع تجمد واجهة المستخدم أثناء الفحص""" global is_checking, available_count, unavailable_count for item in tree.get_children(): tree.delete(item) n = max(1, min(1000, int(count_var.get()))) generated = [] while len(generated) < n: u = make_username() if valid_username(u) and u not in generated: generated.append(u) available_count = 0 unavailable_count = 0 total_var.set(str(n)) available_var.set("0") unavailable_var.set("0") # استخدام ThreadPoolExecutor للسرعة الكبيرة في الفحص التلقائي with ThreadPoolExecutor(max_workers=5) as executor: executor.map(process_single_user, generated) is_checking = False btn.config(state="normal", text="⚡ بدء الفحص التلقائي") def start_generation(): global is_checking if is_checking: return is_checking = True btn.config(state="disabled", text="⏳ جاري الفحص التلقائي...") # إطلاق العمل في الخيط الخلفي threading.Thread(target=run_auto_diagnostics_thread, daemon=True).start() def copy_selected(): selected = tree.selection() if not selected: return value = tree.item(selected[0], "values")[0] root.clipboard_clear() root.clipboard_append(value) root.update() # ===================================================== # ✨ عناصر التحكم والأزرار ✨ # ===================================================== btn = tk.Button( left, text="⚡ بدء الفحص التلقائي", command=start_generation, bg=PINK, fg="white", activebackground="#ff4faf", activeforeground="white", relief="flat", font=("Arial", 12, "bold"), cursor="hand2" ) btn.pack(fill="x", padx=24, pady=(20, 10), ipady=9) copy_btn = tk.Button( left, text="نسخ المحدد", command=copy_selected, bg=PANEL2, fg=TEXT, activebackground="#172943", relief="flat", font=("Arial", 10) ) copy_btn.pack(fill="x", padx=24, ipady=7) note = tk.Label( left, text="ملاحظة: نظام الفحص التلقائي يفحص مباشرة عبر طلبات الشبكة ويحفظ المتاح تلقائياً في ملف available_users.txt", wraplength=360, justify="left", bg=PANEL, fg=MUTED, font=("Arial", 9) ) note.pack(anchor="w", padx=24, pady=18) # جدول عرض النتائج tk.Label( right, text="نتائج الفحص التلقائي الحقيقي", font=("Arial", 15, "bold"), bg=PANEL, fg=TEXT ).pack(anchor="w", padx=22, pady=(20, 8)) columns = ("username", "status", "action") tree = ttk.Treeview(right, columns=columns, show="headings", height=19) tree.heading("username", text="اسم المستخدم") tree.heading("status", text="الحالة التلقائية") tree.heading("action", text="إجراء") tree.column("username", width=190) tree.column("status", width=180) tree.column("action", width=90) tree.pack(fill="both", expand=True, padx=22, pady=(0, 14)) tree.tag_configure("ok", foreground=GREEN) tree.tag_configure("no", foreground=RED) # صندوق الإحصائيات stats = tk.Frame(right, bg=PANEL) stats.pack(fill="x", padx=22, pady=(0, 22)) total_var = tk.StringVar(value="0") available_var = tk.StringVar(value="0") unavailable_var = tk.StringVar(value="0") for label, var in [ ("الإجمالي", total_var), ("متاح تلقائياً", available_var), ("غير متاح", unavailable_var) ]: box = tk.Frame(stats, bg=PANEL2) box.pack(side="left", fill="x", expand=True, padx=4) tk.Label( box, textvariable=var, font=("Arial", 17, "bold"), bg=PANEL2, fg=TEXT ).pack(pady=(10, 0)) tk.Label( box, text=label, font=("Arial", 9), bg=PANEL2, fg=MUTED ).pack(pady=(0, 10)) root.mainloop()
