# ============================================ # TikTok 3-Character Username Checker # يعمل على Google Colab # ============================================ import requests import random import string import time import re # عدد الثلاثيات التي تريد فحصها NUMBER_TO_CHECK = 100 # ثانيتان بين كل طلب DELAY = 2 # حروف + أرقام CHARACTERS = string.ascii_lowercase + string.digits def generate_username(): return ''.join(random.choices(CHARACTERS, k=3)) def check_username(username): url = f"https://www.tiktok.com/@{username}" headers = { "User-Agent": ( "Mozilla/5.0 (iPhone; CPU iPhone OS 17_0 like Mac OS X) " "AppleWebKit/605.1.15 " "(KHTML, like Gecko) " "Version/17.0 Mobile/15E148 Safari/604.1" ), "Accept-Language": "en-US,en;q=0.9" } try: response = requests.get( url, headers=headers, timeout=15, allow_redirects=True ) text = response.text.lower() # TikTok رفض/حدد الطلبات if response.status_code in (403, 429): return "BLOCKED" # بعض صفحات TikTok ترجع 404 if response.status_code == 404: return "NOT_FOUND" # محاولة التعرف على صفحة الحساب غير الموجودة not_found_phrases = [ "couldn't find this account", "couldn’t find this account", "account doesn't exist", "account does not exist" ] if any(x in text for x in not_found_phrases): return "NOT_FOUND" # إذا ظهرت صفحة حساب عادية if response.status_code == 200: return "FOUND_OR_UNKNOWN" return f"UNKNOWN_{response.status_code}" except requests.exceptions.Timeout: return "TIMEOUT" except requests.exceptions.RequestException: return "ERROR" # ============================================ # بدء الفحص # ============================================ checked = set() possible = [] print("=" * 50) print(" TikTok 3-Character Username Checker") print("=" * 50) print() for number in range(1, NUMBER_TO_CHECK + 1): # منع التكرار while True: username = generate_username() if username not in checked: checked.add(username) break print( f"[{number}/{NUMBER_TO_CHECK}] " f"@{username} -> ", end="", flush=True ) result = check_username(username) print(result) if result == "NOT_FOUND": print(f"⭐ مرشح يحتاج تحقق: @{username}") possible.append(username) # إذا TikTok بدأ يمنع الطلبات، نوقف if result == "BLOCKED": print() print("⚠️ TikTok حدّد الطلبات.") print("تم إيقاف الفحص بدل محاولة تجاوز الحماية.") break time.sleep(DELAY) # ============================================ # عرض النتائج # ============================================ print() print("=" * 50) print("النتائج") print("=" * 50) if possible: for username in possible: print("@" + username) # حفظ النتائج with open( "possible_usernames.txt", "w", encoding="utf-8" ) as file: for username in possible: file.write("@" + username + "\n") print() print("تم حفظ النتائج في:") print("possible_usernames.txt") else: print("لم يظهر أي ثلاثي كـ NOT_FOUND.") print() print("مهم: NOT_FOUND لا يعني أن اليوزر متاح للحجز 100%.") print("التأكيد النهائي يكون من داخل TikTok.")
