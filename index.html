from flask import Flask, render_template, request, jsonify, send_file import random import string import os app = Flask(__name__) RESULTS_DIR = "results" RESULTS_FILE = os.path.join(RESULTS_DIR, "available.txt") os.makedirs(RESULTS_DIR, exist_ok=True) def generate_username(length, mode="mixed"): if mode == "letters": chars = string.ascii_lowercase elif mode == "numbers": chars = string.digits else: chars = string.ascii_lowercase + string.digits return "".join(random.choice(chars) for _ in range(length)) @app.route("/") def index(): return render_template("index.html") @app.post("/generate") def generate(): data = request.get_json(silent=True) or {} try: length = int(data.get("length", 4)) amount = int(data.get("amount", 100)) except (TypeError, ValueError): return jsonify({"error": "قيم غير صحيحة"}), 400 length = max(3, min(length, 5)) amount = max(1, min(amount, 5000)) mode = data.get("mode", "mixed") usernames = [] used = set() while len(usernames) < amount: username = generate_username(length, mode) if username not in used: used.add(username) usernames.append(username) return jsonify({ "success": True, "usernames": usernames }) @app.post("/save") def save(): data = request.get_json(silent=True) or {} username = str(data.get("username", "")).strip().lower() allowed = string.ascii_lowercase + string.digits + "_." if not username or len(username) > 24: return jsonify({"success": False}), 400 if any(c not in allowed for c in username): return jsonify({"success": False}), 400 with open(RESULTS_FILE, "a", encoding="utf-8") as f: f.write(username + "\n") return jsonify({"success": True}) @app.get("/download") def download(): if not os.path.exists(RESULTS_FILE): open(RESULTS_FILE, "w", encoding="utf-8").close() return send_file( RESULTS_FILE, as_attachment=True, download_name="available.txt" ) if __name__ == "__main__": app.run( host="0.0.0.0", port=5000, debug=False )
