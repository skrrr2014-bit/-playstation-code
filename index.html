import requests import time import re from pathlib import Path # ============================== # TikTok Username Auto Checker # ============================== INPUT_FILE = "usernames.txt" OUTPUT_FILE = "results.txt" DELAY = 2 # ثانيتان بين كل فحص def clean_username(username): username = username.strip() username = re.sub( r"^https?://(www\.)?tiktok\.com/@?", "", username, flags=re.I ) username = username.lstrip("@") username = username.split("?")[0] username = username.split("/")[0] return username.strip() def valid_username(username): return bool( re.fullmatch(r"[A-Za-z0-9._]{2,24}", username) ) def check_username(username): url = f"https://www.tiktok.com/@{username}" headers = { "User-Agent": ( "Mozilla/5.0 (Windows NT 10.0; Win64; x64) " "AppleWebKit/537.36 " "(KHTML, like Gecko) " "Chrome/140.0 Safari/537.36" ) } try: response = requests.get( url, headers=headers, timeout=15, allow_redirects=True ) # هذه مجرد إشارة لحالة الصفحة if response.status_code == 404: return "NOT FOUND" if response.status_code == 200: return "FOUND" if response.status_code in (403, 429): return "RATE LIMITED" return f"UNKNOWN ({response.status_code})" except requests.exceptions.Timeout: return "TIMEOUT" except requests.exceptions.RequestException: return "ERROR" def main(): input_path = Path(INPUT_FILE) if not input_path.exists(): print(f"لم يتم العثور على ملف {INPUT_FILE}") print("أنشئ ملفًا بهذا الاسم وضع كل يوزر في سطر.") return raw = input_path.read_text( encoding="utf-8", errors="ignore" ) usernames = [] for line in raw.splitlines(): username = clean_username(line) if username and username not in usernames: usernames.append(username) if not usernames: print("القائمة فارغة.") return print("=" * 50) print("TikTok Username Auto Checker") print("=" * 50) print(f"عدد اليوزرات: {len(usernames)}") print() results = [] for number, username in enumerate(usernames, 1): print( f"[{number}/{len(usernames)}] " f"@{username}", end=" -> ", flush=True ) if not valid_username(username): status = "INVALID" else: status = check_username(username) print(status) results.append( f"@{username} | {status}" ) # تأخير بين الطلبات if number < len(usernames): time.sleep(DELAY) Path(OUTPUT_FILE).write_text( "\n".join(results), encoding="utf-8" ) print() print("=" * 50) print(f"انتهى الفحص.") print(f"النتائج محفوظة في: {OUTPUT_FILE}") print("=" * 50) if __name__ == "__main__": main()
