using UnityEngine; using UnityEngine.EventSystems; using UnityEngine.UI; public class MobileGameController : MonoBehaviour { [Header("الشخصية")] public CharacterController player; public Transform husband; [Header("الكاميرا")] public Camera gameCamera; public Transform cameraTarget; [Header("الحركة")] public float moveSpeed = 2.5f; public float rotationSpeed = 8f; [Header("الزوج")] public float husbandSpeed = 1.8f; public float followDistance = 2.5f; [Header("واجهة الجوال")] public Joystick movementJoystick; public Text messageText; [Header("الكاميرا اللمسية")] public float cameraSensitivity = 0.15f; private float yaw = 0f; private float pitch = 10f; private Vector2 lastTouch; private bool rotatingCamera; void Start() { if (messageText != null) messageText.text = ""; yaw = cameraTarget.eulerAngles.y; } void Update() { MovePlayer(); MoveHusband(); TouchCamera(); UpdateCamera(); } // ========================= // حركة الشخصية بالجويستك // ========================= void MovePlayer() { if (movementJoystick == null) return; float horizontal = movementJoystick.Horizontal; float vertical = movementJoystick.Vertical; Vector3 forward = gameCamera.transform.forward; Vector3 right = gameCamera.transform.right; forward.y = 0; right.y = 0; forward.Normalize(); right.Normalize(); Vector3 movement = forward * vertical + right * horizontal; if (movement.magnitude > 1f) movement.Normalize(); player.SimpleMove( movement * moveSpeed ); if (movement.magnitude > 0.1f) { Quaternion targetRotation = Quaternion.LookRotation(movement); player.transform.rotation = Quaternion.Slerp( player.transform.rotation, targetRotation, rotationSpeed * Time.deltaTime ); } } // ========================= // الزوج // ========================= void MoveHusband() { if (husband == null) return; Vector3 direction = player.transform.position - husband.position; direction.y = 0; float distance = direction.magnitude; if (distance > followDistance) { Vector3 movement = direction.normalized * husbandSpeed * Time.deltaTime; husband.position += movement; if (movement.sqrMagnitude > 0.001f) { Quaternion targetRotation = Quaternion.LookRotation(movement); husband.rotation = Quaternion.Slerp( husband.rotation, targetRotation, 8f * Time.deltaTime ); } } } // ========================= // تحريك الكاميرا بالإصبع // ========================= void TouchCamera() { if (Input.touchCount == 0) return; Touch touch = Input.GetTouch(0); // لا تحرك الكاميرا إذا كان الإصبع على زر/جويستك if (EventSystem.current != null && EventSystem.current.IsPointerOverGameObject(touch.fingerId)) { return; } if (touch.phase == TouchPhase.Began) { lastTouch = touch.position; rotatingCamera = true; } if (touch.phase == TouchPhase.Moved && rotatingCamera) { Vector2 delta = touch.position - lastTouch; yaw += delta.x * cameraSensitivity; pitch -= delta.y * cameraSensitivity; pitch = Mathf.Clamp( pitch, -10f, 45f ); lastTouch = touch.position; } if (touch.phase == TouchPhase.Ended || touch.phase == TouchPhase.Canceled) { rotatingCamera = false; } } // ========================= // الكاميرا // ========================= void UpdateCamera() { if (gameCamera == null || cameraTarget == null) return; Quaternion rotation = Quaternion.Euler( pitch, yaw, 0 ); Vector3 offset = rotation * new Vector3( 0f, 1.6f, -4f ); Vector3 wantedPosition = cameraTarget.position + offset; gameCamera.transform.position = Vector3.Lerp( gameCamera.transform.position, wantedPosition, 12f * Time.deltaTime ); gameCamera.transform.LookAt( cameraTarget.position + Vector3.up * 1.1f ); } // ========================= // الأزرار // ========================= public void Talk() { ShowMessage("💬 تحدثتما معًا."); } public void Hug() { ShowMessage("🤗 عناق لطيف."); } public void Dance() { ShowMessage("💃 رقصتما معًا."); } public void Sit() { ShowMessage("🛋️ جلستما معًا."); } void ShowMessage(string text) { if (messageText == null) return; messageText.text = text; CancelInvoke(nameof(ClearMessage)); Invoke( nameof(ClearMessage), 2f ); } void ClearMessage() { if (messageText != null) messageText.text = ""; } }
