import requests import random import string import time # ========================= # الإعدادات # ========================= NUMBER_TO_CHECK = 100 DELAY = 2 # ثلاثيات فقط: حروف + أرقام CHARS = string.ascii_lowercase + string.digits def generate_username(): return ''.join(random.choices(CHARS, k=3)) def check_username(username): url = f"https://www.tiktok.com/@{username}" headers = { "User-Agent": ( "Mozilla/5.0 (iPhone; CPU iPhone OS 17_0 like Mac OS X) " "AppleWebKit/605.1.15 " "(KHTML, like Gecko) " "Version/17.0 Mobile/15E148 Safari/604.1" ) } try: response = requests.get( url, headers=headers, timeout=15, allow_redirects=True ) if response.status_code == 404: return "NOT FOUND" if response.status_code == 200: return "FOUND" if response.status_code == 429: return "RATE LIMITED" if response.status_code == 403: return "BLOCKED" return f"UNKNOWN ({response.status_code})" except requests.exceptions.RequestException: return "ERROR" def main(): checked = set() results = [] print("=" * 45) print(" TikTok Random 3-Letter Username Checker") print("=" * 45) print() while len(checked) < NUMBER_TO_CHECK: username = generate_username() if username in checked: continue checked.add(username) print( f"[{len(checked)}/{NUMBER_TO_CHECK}] " f"@{username} -> ", end="", flush=True ) status = check_username(username) print(status) if status == "NOT FOUND": print(f"⭐ مرشح غير موجود: @{username}") results.append(username) if status == "RATE LIMITED": print("تم الوصول إلى حد الطلبات. أوقف الفحص مؤقتًا.") break time.sleep(DELAY) with open("possible_usernames.txt", "w", encoding="utf-8") as f: for username in results: f.write("@" + username + "\n") print() print("=" * 45) print("انتهى الفحص") print(f"تم العثور على {len(results)} مرشح غير موجود.") print("النتائج محفوظة في possible_usernames.txt") print("=" * 45) if __name__ == "__main__": main()
